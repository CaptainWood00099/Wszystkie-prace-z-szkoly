class samochod{
    constructor(marka, model, rok_produkcji){
    this.marka=marka;
    this.model=model;
    this.rok_produkcji=rok_produkcji;
}
}

samochod.spalanie=function(l,km){
    spalanie=(l/km)*100;
    return spalanie;
}

const samochod1= new samochod("fiat", "panda", "2024");
const samochod2= new samochod("toyota", "yaris", "2021");
const samochod3= new samochod("BMW", "350i", "2007");
samochod1.spalanie(10,100)
samochod2.spalanie(10,100)
samochod3.spalanie(10,100)
